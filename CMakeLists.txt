cmake_minimum_required(VERSION 3.16)
project(RideFlow VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(rideflow src/engine.cpp)
target_include_directories(rideflow PUBLIC include)
target_link_libraries(rideflow PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(rideflow PRIVATE /W4)
else()
    target_compile_options(rideflow PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(rideflow_demo examples/demo.cpp)
target_link_libraries(rideflow_demo PRIVATE rideflow)
include(CTest)
if(BUILD_TESTING)
    add_executable(rideflow_tests tests/engine_tests.cpp)
    target_link_libraries(rideflow_tests PRIVATE rideflow)
    add_test(NAME rideflow_tests COMMAND rideflow_tests)
    set_tests_properties(rideflow_tests PROPERTIES TIMEOUT 60)
endif()
